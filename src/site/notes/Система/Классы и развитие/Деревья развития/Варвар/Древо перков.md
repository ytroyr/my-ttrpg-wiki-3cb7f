<script>
  setInterval(() => {
    const svg = document.querySelector('svg[id^="mermaid"]');
    if (!svg || svg.dataset.zoomReady) return;
    svg.dataset.zoomReady = "true";

    const box = svg.parentElement;
    box.style.position = "relative";
    box.style.overflow = "hidden";
    svg.style.transformOrigin = "center center";

    let scale = 1, panX = 0, panY = 0, drag = false, sx, sy;

    const update = () => {
      svg.style.transform = `translate(${panX}px, ${panY}px) scale(${scale})`;
    };

    const bar = document.createElement('div');
    bar.className = 'm-zoom-bar';
    bar.innerHTML = '<button type="button" class="m-zoom-btn">+</button><button type="button" class="m-zoom-btn">−</button><button type="button" class="m-zoom-btn">⟲</button>';
    box.appendChild(bar);

    const btns = bar.querySelectorAll('button');
    btns[0].onclick = (e) => { e.preventDefault(); e.stopPropagation(); scale = Math.min(scale + 0.3, 6); update(); };
    btns[1].onclick = (e) => { e.preventDefault(); e.stopPropagation(); scale = Math.max(scale - 0.3, 0.4); update(); };
    btns[2].onclick = (e) => { e.preventDefault(); e.stopPropagation(); scale = 1; panX = 0; panY = 0; update(); };

    box.addEventListener('wheel', (e) => {
      e.preventDefault();
      scale = Math.min(Math.max(scale + (e.deltaY < 0 ? 0.2 : -0.2), 0.4), 6);
      update();
    }, { passive: false });

    box.addEventListener('mousedown', (e) => {
      if (e.target.closest('a') || e.target.closest('button')) return;
      drag = true;
      sx = e.clientX - panX;
      sy = e.clientY - panY;
      svg.style.cursor = 'grabbing';
    });
    window.addEventListener('mousemove', (e) => {
      if (!drag) return;
      panX = e.clientX - sx;
      panY = e.clientY - sy;
      update();
    });
    window.addEventListener('mouseup', () => {
      drag = false;
      svg.style.cursor = 'grab';
    });
  }, 400);
</script>
